cat > ~/build_pytorch.sh << 'EOF'
#!/bin/bash
set -e

echo "[1/6] Updating system..."
sudo apt update && sudo apt upgrade -y
sudo apt install -y build-essential git python3.12 python3.12-dev python3.12-venv cmake ninja-build

echo "[2/6] Creating Python venv..."
python3.12 -m venv ~/pytorch_env
source ~/pytorch_env/bin/activate
pip install --upgrade pip

echo "[3/6] Installing CUDA dependencies..."
sudo apt install -y nvidia-cuda-toolkit

echo "[4/6] Cloning PyTorch..."
cd ~
git clone --recursive https://github.com/pytorch/pytorch.git
cd pytorch
git checkout v2.4.1

echo "[5/6] Setting build environment..."
export USE_CUDA=1
export USE_NCCL=0
export USE_FLASH_ATTENTION=0
export MAX_JOBS=4
export TORCH_CUDA_ARCH_LIST="12.0"
export CMAKE_CUDA_ARCHITECTURES="120"

echo "[6/6] Building PyTorch (this takes 20-30 mins)..."
pip install -r requirements.txt
python setup.py develop 2>&1 | tail -20

echo "Build complete. Testing CUDA..."
python -c "import torch; print(f'CUDA: {torch.cuda.is_available()}'); print(f'Device: {torch.cuda.get_device_name(0) if torch.cuda.is_available() else \"CPU\"}')"
EOF

chmod +x ~/build_pytorch.sh
~/build_pytorch.sh


sudo apt remove -y cuda-toolkit
sudo apt autoremove -y

# Add NVIDIA CUDA repo for 12.9
wget https://developer.download.nvidia.com/compute/cuda/repos/ubuntu2404/x86_64/cuda-ubuntu2404.pin
sudo mv cuda-ubuntu2404.pin /etc/apt/preferences.d/cuda-repository-pin-600
sudo apt-key adv --fetch-keys https://developer.download.nvidia.com/compute/cuda/repos/ubuntu2404/x86_64/3bf863cc.pub
sudo add-apt-repository "deb https://developer.download.nvidia.com/compute/cuda/repos/ubuntu2404/x86_64/ /"

sudo apt update
sudo apt install -y cuda-toolkit-12-9

# Set paths
export PATH=/usr/local/cuda-12.9/bin:$PATH
export LD_LIBRARY_PATH=/usr/local/cuda-12.9/lib64:$LD_LIBRARY_PATH
export CUDA_HOME=/usr/local/cuda-12.9

# Verify
nvcc --version
